SetupWorld_Begin
	PutMessage	SetupWorld	
	// Start showing init update
	SetValue		[#ForceUpdate: TRUE]
	SetValue		[#ForceState: #initializing]
	SetValue		[#Scene: "World"]
	UpdateTagsInitializingGame	x	x
	// Update shockwave trophies
	UpdateSWTrophies	x	x
	// TrackInfo
	SetValue		[#Track: "World{:World}"]
	UpdateTrackInfo
	// Setup textures & world
	Event	SetupTexturesWorld
	Event	SetupWorld{:World}
	// Setup objects & pickups
	Event	SetupObjects
	Event	SetupDebris
	Event	SetupPickups
	Event	SetupWorldMarker
	// Clone objects, pickups and markers
	CloneModel	World{:world}	[#SourceMember: "Objects", #lod: FALSE, #delete: TRUE]
	CloneModel	World{:world}	[#SourceMember: "Debris",   #lod: FALSE, #delete: TRUE]
	CloneModel	World{:world}	[#SourceMember: "Pickups", #lod: FALSE, #delete: TRUE]
	CloneModel	World{:world}	[#SourceMember: "WorldMarker", #lod: FALSE, #delete: TRUE]	
	// Build sprite & add values
	//ResetSprites	x	x
	CreateSprite	World{:World}	[#channel: 1, #default3dScripts: TRUE]
	AddValue		[#name: "Member", #value: member("World{:World}")]
	AddValue		[#name: "Sprite", #value: sprite(1)]
	// Add world & vehicle manager
	// The game will start when world is initialized and the necessary vehicles have been downloaded
	// If offline: imported when accessing the garage OR when starting a challenge
	// If online: hold while downloading then import when accessing the garage or hold while downloading then import when starting a challenge
	AddScript	#Game	[#script: "[PS] World Manager", #data: [#member: "World{:World}"]]	
	AddScript	#System	[#script: "[PS] Vehicle Manager", #data: [#member: "World{:World}"]]	
SetupWorld_End

WorldGeneralFinish_Begin
	PutMessage	--> WorldGeneralFinish	
	// Traffic
	Evaluate	{:WorldTraffic}=TRUE	[#event: "WorldTrafficSetup"]
	// Ambient style	
	SetAmbientStyle	World{:World}	[#style: #World{:World}{:Setting}]		
	// Restore some shaders
	SetShader	World{:World}	[#shader: "*WorldMarker", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "Pickups_Material", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "Decals_Material", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "ImpactSmoke_Material", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "ImpactBurst_Material", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "ImpactSparks_Material", #emissive: rgb(255,255,255)]
	SetShader	World{:World}	[#shader: "ImpactDebris_Material", #emissive: rgb(128,128,128)]
	SetShader	World{:World}	[#shader: "Wireframe_Material", #emissive: rgb(255,255,255)]
	// SPECIAL EFFECTS: Motion blur
	AddScript	#system	[#script: "[PS] Motion Blur", #duplicate: FALSE, #data: [#member: "World{:World}", #camera: "Player1_Camera", #passes: 7]]
	AddScript	#system	[#script: "[PS] Bloom", #duplicate: FALSE, #data: [#member: "World{:World}", #camera: "Player1_Camera", #template: #World{:World}{:Setting}, #skybox: TRUE]]
	SetAntiAliasing	{:Sprite}	[#mode: gGame.AntiAliasing]	
	// FPS Counter
	AddScript		[#script: "[PS] FPSCounter", #data: [:]] 
	// set index
	// WORLD SPECIAL EFFECTS & GENERAL EFFECTS
	Event	SetupWorld{:World}SpecialEffects
	Event	SetupWorld{:World}Effects{:Effects}
	// Always Pause
	PauseGame	x	[#particles: FALSE, #animations: FALSE]
	// Menu OR direct start
	Evaluate	{:MenuStart}=FALSE	[#event: "FadeStartGame"]
	Evaluate	{:MenuStart}=TRUE	[#event: "FadeStartIntro"]	
	// update collision reports
	CreateCollisionReport		[#nameA: gGame.vehicles[1].vehicle.proxy.name, #nameB: #all, #collisionDamage: TRUE]
	// show sprite
	SetSprite	{:Sprite}
	// Stop updating
	SetValue		[#ForceUpdate: FALSE]	
WorldGeneralFinish_End

DeleteVehicleBuildCar_Begin
	PutMessage	--> DeleteVehicleBuildCar
	// Store value
	SetValue		[#PreviousVehicle: "{:Vehicle}"]
	Evaluate	"{:PreviousVehicle}"="Helicopter"	[#handler: "RestoreAndShowCurrentMiniMap"]
	// Delete vehicle
	DeleteVehicle		[#index: 1]
	// Remove all vehicle markers
	RemoveMarker	World{:World}	[#name: "{:World}BoatToCar_Marker"]
	RemoveMarker	World{:World}	[#name: "{:World}HelicopterToCar_Marker"]
	// Add marker to switch back to boat & heli
	AddMarker	World{:World}	[#position:  "{:World}CarToBoat_Marker", #Type: #SwitchToBoat, #active: FALSE]
	AddMarker	World{:World}	[#position:  "{:World}CarToHelicopter_Marker", #Type: #SwitchToHelicopter, #active: FALSE]
	// Start dummy to build car & disable it	
	SetValue	gGame.WorldStartDummy	[#{:World}: "{:World}CarFrom{:PreviousVehicle}_Marker"] 
	DisableCurrentMarker	x	x
	// Create script and add at correct position in scriptList
	AddCarScript	x	x
  	// Update interface
	UpdateWeaponsData	x	x
	// Set camera target
	SetValue	gGame.Camera[1]	[#target: gGame.Vehicles[1].general.tag]
	SetValue	gGame.Camera[1]	[#DefaultSet: #{:Car}]
	UpdateCameraSet	#{:Car}
	// Stop ambient, water & clouds (if the are playing or not)
	StopSound	Ambient{:World}_SFX
	StopSound	AmbientWater_SFX
	StopSound	AmbientClouds_SFX
	// Play default ambient (specific per world)
	PlaySound	Ambient{:World}_SFX	[#priority: #top, #volume: 1.0, #loopcount: 0]
	FadeSound	Ambient{:World}_SFX	[#fadeTime: 750, #fadeOut: FALSE])  
	// Update world dummy (reset position)
	SetValue	gGame.WorldStartDummy	[#{:World}: "SafeHouse{:World}_Marker"] 
	// Set value
	SetValue		[#Vehicle: "Car"]
	SetValue		[#PreviousVehicle: #none]
	// Fade
	Fade	World{:World}	[#time: 500]	
DeleteVehicleBuildCar_End
