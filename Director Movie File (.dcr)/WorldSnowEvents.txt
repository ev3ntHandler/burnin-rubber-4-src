SetupWorldSnow_Begin
	PutMessage	--> SetupWorldSnow
	// ModelList
	SetValue		[#WorldModelList: {:WorldModels}.duplicate()]
	// Additional textures & shader sets
	Evaluate	"{:Setting}"="Night"	[#Event: "WorldSnowAdditiveTextures"]	
	// Skybox
	Event	SkyboxInitialization
	SetSkybox	World{:World}	[#skybox: "SkyboxSnow"]
SetupWorldSnow_End

LandMarkSnow_Begin
	PutMessage	--> LandMarkSnow
	// Sound
	PlaySound	SpecialPickUp_SFX	[#override: TRUE]
	// Text textures
	CreateTextTexture	Interface	[#tag: #InterfaceStatusHeaderText, #style: #StatusTitle, #text: #LandmarkInformationTitle]
	CreateTextTexture	Interface	[#tag: #InterfaceStatusBodyText, #style: #StatusBody, #text: #LandmarkInformation]
	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World{:World}"]
	// Delay next event
	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
	// Do it
	SetModel	World{:World}	[#model: "04_12_landmarksnow_1_object", #resource: "landmarksnowdestroyed_object", #shader: "WorldSnowMisc_Material"] 
	AddToList	gGame.UPLandMarks	[#value: "LandMarkSnow_Object", #override: TRUE]
	SaveData	x	x
	CheckAllLandMarks	x	x
LandMarkSnow_End
