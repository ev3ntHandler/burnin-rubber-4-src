EnterShop_Begin
	PutMessage	EnterShop
	// Sound
	PlaySound	Marker_SFX	[#override: TRUE]
	// Hide Status if necessary
	Event	HideStatus
	// no input
	SetValue		[#input: FALSE]
	// in shop
	SetValue		[#inShop: TRUE]
	// Hold car & Show/Hide interface
	HoldCarForShop	x	x
	Event	HideCurrentInterface
	// set camera
	UpdateCameraSet	#MimicHard
	SetValue	gGame.Camera[1]	[#state: #mimic]
	SetValue	gGame.Camera[1]	[#target: "{:CurrentMarker}_Camera"]
	// (Un)bind esc
	Event	UnbindWorldKeys
	// Flash Fade
	Fade	World{:World}	[#time: 250, #type: #out]
EnterShop_End
 
ExitShop_Begin
	PutMessage	ExitShop
	// Hold car & Show/Hide interface
	SetVisible	Interface	[#model: "InterfaceOverlayPause", #state: TRUE]
	AddToFader	Interface	[#model: "InterfaceOverlayPause", #fade: 100, #start: 0, #time: 600.0]
	DelayEvent		[#delay: 750, #event: "ClosingUpShop"]
	SetVisible	Interface	[#model: ["{:CurrentShop}", "Misc"], #state: FALSE]
	ReleaseCarFromShop	x	x
	ResetCurrentShop	x	x
	// Camera
	UpdateCameraSet	#{:Car}
	SetValue	gGame.Camera[1]	[#state: #follow]
	SetValue	gGame.Camera[1]	[#target: gGame.Vehicles[1].general.tag]
	SetValue		[#CurrentMarker: #none]
	// Bind Keys
	Event	BindWorldKeys
	// Force Save
	SaveData	x	x
	// Flash Fade
	Fade	World{:World}	[#time: 250, #type: #out]	
ExitShop_End

SetupSafeHouse_Begin
	PutMessage	SetupSafeHouse (FADE)
	// no input
	SetValue		[#input: FALSE]
	//Show/Hide interface
	Event	HideCurrentInterface
	// Fade
	Fade	World{:World}	[#time: 250, #type: #in, #event: "EnterSafeHouse"]]
SetupSafeHouse_End	

EnterSafeHouse_Begin
	PutMessage	EnterSafeHouse
	// Hold car
	HoldCarForShop	x	x
	// set camera
	UpdateCameraSet	#MimicHard
	SetValue	gGame.Camera[1]	[#state: #mimic]
	// (Un)bind esc
	Event	UnbindWorldKeys	
	// Value
	SetValue		[#currentShop: #Safehouse]
	// Show
	SetVisible	Interface	[#model: ["Safehouse", "Misc"], #state: TRUE]
	SetVisible	Interface	[#model: ["Tickets", "Money"], #state: TRUE, #parent: "Misc"]
	SetVisible	Interface	[#model: ["TotalCost", "Pause", "CenterButton", "SafehouseTicketBig"], #state: FALSE]
	// Set buttons
	SetModel	Interface	[#model: "LeftButton", #resource: "ExitButton", #shader: "GreyInterface_Material"]
	SetModel	Interface	[#model: "RightButton", #resource: "SelectButton", #shader: "GreyInterface_Material"]
	// Create buttons
	CreateButton	Interface	[#model: "LeftButton", #event: "RestoreVehicleSafeHouse", #effect: #shaderRollover, #shaders: ["GreyInterface_Material", "Interface_Material"]]
	CreateButton	Interface	[#model: "RightButton", #event: "FadeResetSafehouse", #effect: #shaderRollover, #shaders: ["GreyInterface_Material", "Interface_Material"]]
	// show sprite
	SetSprite	{:Sprite}	[#visible:FALSE]
	// Evaluate Safehouse
	Event	EvaluateSafeHouse
EnterSafeHouse_End	

EvaluateSafeHouse_Begin
	PutMessage	EvaluateSafeHouse
	// Clear Triggers
	SetValue	gData	[#Triggers: []]
	// What to do...
	Evaluate	gGame.VehiclesLoaded.set2=#done	[#event: "ContinueEnterSafeHouse"]
	Evaluate	gGame.VehiclesLoaded.set2=#ready	[#event: "ImportSet2SafeHouse"]
	Evaluate	gGame.VehiclesLoaded.set2=#loading	[#event: "LoadSet2SafeHouse"]		
EvaluateSafeHouse_End

LoadSet2SafeHouse_Begin
	PutMessage	LoadSet2SafeHouse
	SetupProgressLoader	x	x
	AddTrigger	gGame.VehiclesLoaded.set2=#loading	[#handler: "UpdateProgressLoader", #runOnce: 0, #frameFreq: 1]
	AddTrigger	gGame.VehiclesLoaded.set2=#ready	[#event: "EvaluateSafeHouse", #handler: "ResetProgressLoader", #runOnce: 1, #frameFreq: 1]
LoadSet2SafeHouse_End

ImportSet2SafeHouse_Begin
	PutMessage	SafeHouseImportSet2AndContinue
	// Start showing init update
	SetValue		[#ForceUpdate: TRUE]
	SetValue		[#ForceState: #loading]
	Event	ImportVehicleSet2
	SetValue		[#ForceUpdate: FALSE]
	Event	EvaluateSafeHouse	
ImportSet2SafeHouse_End

ContinueEnterSafeHouse_Begin
	PutMessage	--> ContinueSetupSafeHouse
	SetupSafehouse	x	x
	Fade	World{:World}	[#time: 500, #type: #out]
	SetSprite	{:Sprite}
ContinueEnterSafeHouse_End

RestoreVehicleSafeHouse_Begin

RestoreVehicleSafeHouse_End

FadeResetSafehouse_Begin
	PutMessage	--> FadeResetSafehouse

FadeResetSafehouse_End

EvaluateTrackInitialization_Begin
	PutMessage	EvaluateTrackInitialization
	// Clear Triggers
	SetValue	gData	[#Triggers: []]
	// What to do...
	Evaluate	gGame.VehiclesLoaded.set2=#done;gGame.VehiclesLoaded.set3=#done	[#event: "TrackInitialization"]
	Evaluate	gGame.VehiclesLoaded.set2=#done;gGame.VehiclesLoaded.set3=#ready	[#event: "ImportSet3TrackInitialization"]
	Evaluate	gGame.VehiclesLoaded.set2=#done;gGame.VehiclesLoaded.set3=#loading	[#event: "LoadSet3TrackInitialization"]
	Evaluate	gGame.VehiclesLoaded.set2=#ready	[#event: "ImportSet2TrackInitialization"]
	Evaluate	gGame.VehiclesLoaded.set2=#loading	[#event: "LoadSet2TrackInitialization"]	
EvaluateTrackInitialization_End

	// Sound
	PlaySound	Status_SFX	[#override: TRUE]

	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World"]

	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
WeaponShopUnavailable_End
