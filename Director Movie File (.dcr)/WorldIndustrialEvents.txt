SetupWorldIndustrial_Begin
	PutMessage	--> SetupWorldIndustrial
	// ModelList
	SetValue		[#WorldModelList: {:WorldModels}.duplicate()]
	AddToList	gGame.WorldModelList	[#value: "Street2"]
	AddToList	gGame.WorldModelList	[#value: "Buildings2"]
	AddToList	gGame.WorldModelList	[#value: "Sea"]
	// Additional default textures
	CreateTexture	World{:World}	[#texture: "WorldSea{:Setting}_Texture"]
	CreateTexture	World{:World}	[#texture: "WorldWaves{:Setting}_Texture"]
	CreateTexture	World{:World}	[#texture: "WorldIndustrialSeaLightmap_Texture"]
	SetShader	World{:World}	[#shader: "WorldIndustrialSea_Material", #texture1: "WorldSea{:Setting}_Texture", #texture2: "WorldIndustrialSeaLightmap_Texture"]
	SetShader	World{:World}	[#shader: "WorldIndustrialWaves_Material", #texture1: "WorldWaves{:Setting}_Texture"]
	// Additional textures & shader sets
	CreateTexture	World{:World}	[#texture: "WorldIndustrialBuildings_Texture"]
	CreateTexture	World{:World}	[#texture: "WorldIndustrialStreet_Texture"]
	SetShader	World{:World}	[#shader: "WorldIndustrialBuildings2_Material", #texture1: "WorldIndustrialBuildings_Texture", #texture2: "WorldIndustrialBuildingsLightmap_Texture"]
	SetShader	World{:World}	[#shader: "WorldIndustrialStreet2_Material", #texture1: "WorldIndustrialStreet_Texture", #texture2: "WorldIndustrialStreetLightmap_Texture"]
	Evaluate	"{:Setting}"="Night"	[#Event: "WorldIndustrialAdditiveTextures"]
	// Skybox
	Event	SkyboxInitialization
	Event	WorldIndustrialSkybox{:Setting}
	//SetSkybox	World{:World}	[#skybox: "SkyboxIndustrial"]
SetupWorldIndustrial_End

NuclearDetonatorIndustrial1_Begin
	PutMessage	--> NuclearDetonatorIndustrial1
	// Std detonator
	Event	NuclearDetonator
	// set camera
	SetValue	gGame.Camera[1]	[#state: #mimic]
	SetValue	gGame.Camera[1]	[#target: "NuclearDetonatorIndustrial1_Camera"]

LandMarkIndustrial_Begin
	PutMessage	--> LandMarkIndustrial
	// Sound
	PlaySound	SpecialPickUp_SFX	[#override: TRUE]
	// Text textures
	CreateTextTexture	Interface	[#tag: #InterfaceStatusHeaderText, #style: #StatusTitle, #text: #LandmarkInformationTitle]
	CreateTextTexture	Interface	[#tag: #InterfaceStatusBodyText, #style: #StatusBody, #text: #LandmarkInformation]
	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World"]
	// Delay next event
	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
	// Do it
	SetModel	World{:World}	[#model: "06_08_landmarkIndustrial_1_object", #resource: "landmarkIndustrialdestroyed_object", #shader: "WorldIndustrialMisc_Material"] 
	AddToList	gGame.UPLandMarks	[#value: "LandMarkIndustrial_Object", #override: TRUE]
	SaveData	x	x
	CheckAllLandMarks	x	x
LandMarkIndustrial_End
