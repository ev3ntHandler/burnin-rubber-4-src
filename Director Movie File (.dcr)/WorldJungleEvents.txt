SetupWorldJungle_Begin
	PutMessage	--> SetupWorldJungle
	// ModelList
	SetValue		[#WorldModelList: {:WorldModels}.duplicate()]
	AddToList	gGame.WorldModelList	[#value: "Sea"]
	AddToList	gGame.WorldModelList	[#value: "Waves"]
	// Additional textures & shader sets
	CreateTexture	World{:World}	[#texture: "WorldSea{:Setting}_Texture"]
	SetShader	World{:World}	[#shader: "WorldJungleSea_Material", #texture1: "WorldSea{:Setting}_Texture"]
	SetShader	World{:World}	[#shader: "WorldJunglePool_Material", #texture1: "WorldSea{:Setting}_Texture"]
	CreateTexture	World{:World}	[#texture: "WorldWaves{:Setting}_Texture"]
	SetShader	World{:World}	[#shader: "WorldJungleWaves_Material", #texture1: "WorldWaves{:Setting}_Texture"]
	Evaluate	"{:Setting}"="Night"	[#Event: "WorldJungleAdditiveTextures"]	
	// Skybox
	Event	SkyboxInitialization
	SetSkybox	World{:World}	[#skybox: "SkyboxJungle"]
SetupWorldJungle_End	

SetupWorldJungleSpecialEffects_Begin
	PutMessage	SetupWorldJungleSpecialEffects	
	// Scroller
	AddToScroller	World{:World}	[#shader: "WorldJungleSea_Material", #position: vector(-0.001, -0.001, 0.0)]
	AddToScroller	World{:World}	[#shader: "WorldJunglePool_Material", #position: vector(-0.001, -0.001, 0.0)]
SetupWorldJungleSpecialEffects_End

	PutMessage	--> NuclearDetonatorJungle1
	// Std detonator
	Event	NuclearDetonator
	// set camera

LandMarkJungle_Begin
	PutMessage	--> LandMarkJungle
	// Sound
	PlaySound	SpecialPickUp_SFX	[#override: TRUE]
	// Text textures
	CreateTextTexture	Interface	[#tag: #InterfaceStatusHeaderText, #style: #StatusTitle, #text: #LandmarkInformationTitle]
	CreateTextTexture	Interface	[#tag: #InterfaceStatusBodyText, #style: #StatusBody, #text: #LandmarkInformation]
	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World{:World}"]
	// Delay next event
	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
	// Do it
	SetModel	World{:World}	[#model: "06_08_landmarkjungle_1_object", #resource: "landmarkjungledestroyed_object", #shader: "WorldJungleMisc_Material"] 
	AddToList	gGame.UPLandMarks	[#value: "LandMarkJungle_Object", #override: TRUE]
	SaveData	x	x
	CheckAllLandMarks	x	x
LandMarkJungle_End
