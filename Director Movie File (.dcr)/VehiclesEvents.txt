//-------------------------------------------------------------------------------------------------------------------------------------------
SetupCar_Begin
	PutMessage	--> Setup{:Vehicle}
	// Textures
	CreateTexture	{:SetupCar}	[#texture: "{:SetupCar}Skin{:SetupSkin}_Texture", #filtering: #lowfiltered]
	CreateTexture	{:SetupCar}	[#texture: "Wheels_Texture", #filtering: #lowfiltered])
	CreateTexture	{:SetupCar}	[#texture: "Reflection_Texture", #filtering: #lowfiltered])
	CreateTexture	{:SetupCar}	[#texture: "FrontLight_Texture", #filtering: #lowfiltered])
	CreateTexture	{:SetupCar}	[#texture: "RearLight_Texture", #filtering: #lowfiltered])
	// Set textures channel 1
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}_Material", #texture1: "{:SetupCar}Skin{:SetupSkin}_Texture", #shininess: 30.0]
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}Glass_Material", #texture1: "{:SetupCar}Skin{:SetupSkin}_Texture", #shininess: 35.0]
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}Wheels_Material", #texture1: "Wheels_Texture", #shininess: 30.0]
	SetShader	{:SetupCar}	[#shader: "FrontLight_Material", #texture1: "FrontLight_Texture"]
	SetShader	{:SetupCar}	[#shader: "RearLight_Material", #texture1: "RearLight_Texture"]
	// Additive shaders light
	BlendShader	{:SetupCar}	[#shader: "FrontLight_Material", #blendMode: #add]
	BlendShader	{:SetupCar}	[#shader: "RearLight_Material", #blendMode: #add]
	// Set textures channel 2
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}_Material", #texture2: "Reflection_Texture"]
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}Glass_Material", #texture2: "Reflection_Texture"]
	// Set blendmodes
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}_Material", #blendFunction1: #multiply, #textureMode2: #reflection, #blendFunction2: #add]
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}Glass_Material", #blendFunction1: #multiply, #textureMode2: #reflection, #blendFunction2: #add]
	SetShader	{:SetupCar}	[#shader: "{:SetupCar}Wheels_Material", #shininess: 0.0]
SetupCar_End
