DesertTextures_Begin
	// Textures
	CreateTexture	DesertAssets	[#texture: "DesertStreet_Texture"]
	CreateTexture	DesertAssets	[#texture: "DesertBuildings_Texture"]
	CreateTexture	DesertAssets	[#texture: "DesertTransparent_Texture", #renderFormat: #rgba8888]
	CreateTexture	DesertAssets	[#texture: "Wall_Texture", #renderFormat: #rgba8888]

	SetShader	DesertAssets	[#shader: "DesertBuildings_Material", #texture1: "DesertBuildings_Texture"]
	SetShader	DesertAssets	[#shader: "DesertTransparent_Material", #texture1: "DesertTransparent_Texture"]

	SetShader	DesertAssets	[#shader: "DesertCollision_Material", #texture1: "DesertTransparent_Texture"]
	SetShader	DesertAssets	[#shader: "WorldMisc_Material", #texture1: "WorldMisc_Texture"]
	
	// Transparent off for opaque
	SetShader	DesertAssets	[#shader: "DesertOpaque_Material", #transparent: FALSE]
	
	// AmbientStyle

	SetShader	DesertAssets	[#shader: "DesertTransparent_Material", #emissive: gData.data.ambientstyle[gGame.trackinfo.ambientStyle].emissive]
