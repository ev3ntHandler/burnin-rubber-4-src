//----------------------------------------------------------------------
// Weapons
//----------------------------------------------------------------------
SetupWeapons_Begin
	CreateTexture	Weapons	[#texture: "Weapons_Texture"]
	CreateTexture	Weapons	[#texture: "RocketFlame_Texture"]
	CreateTexture	Weapons	[#texture: "Effects_Texture"]
	CreateTexture	Weapons	[#texture: "Flame_Texture"]
	SetShader	Weapons	[#shader: "Weapons_Material", #shininess: 60.0]
	BlendShader	Weapons	[#shader: "Effects_Material", #blendMode: #add]
	BlendShader	Weapons	[#shader: "Flame_Material", #blendMode: #add]
SetupWeapons_End

//----------------------------------------------------------------------
// Objects
//----------------------------------------------------------------------
SetupObjects_Begin
	PutMessage	SetupObjects
	ResetWorld	Objects
	CreateTexture	Objects	[#texture: "WorldMisc_Texture"]
	CreateTexture	Objects	[#texture: "WorldMarker_Texture", #renderFormat: #rgba8888]
	CreateTexture	Objects	[#texture: "Arrows_Texture", #renderFormat: #rgba8888]
	CreateTexture	Objects	[#texture: "Crosses_Texture", #renderFormat: #rgba8888]
	CreateTexture	Objects	[#texture: "Finish_Texture", #renderFormat: #rgba8888]
	SetShader	Objects	[#shader: "WorldMisc_Material", #texture1: "WorldMisc_Texture"]
	SetShader	Objects	[#shader: "WorldMarker_Material", #texture1: "WorldMarker_Texture"]
	SetShader	Objects	[#shader: "Arrows_Material", #texture1: "Arrows_Texture"]
	SetShader	Objects	[#shader: "Crosses_Material", #texture1: "Crosses_Texture"]
	SetShader	Objects	[#shader: "Finish_Material", #texture1: "Finish_Texture"]
	AddValue		[#name: "ObjectsInitialized", #value: TRUE]
SetupObjects_End

ResetObjects_Begin
	PutMessage	ResetObjects
	ResetWorld	Objects
ResetObjects_End
