SetupWorldCity_Begin
	PutMessage	--> SetupWorldCity
	// ModelList
	SetValue		[#WorldModelList: {:WorldModels}.duplicate()]
	AddToList	gGame.WorldModelList	[#value: "Sea"]
	AddToList	gGame.WorldModelList	[#value: "Waves"]
	
	// Additional textures & shader sets
	CreateTexture	World{:World}	[#texture: "WorldSea{:Setting}_Texture"]
	CreateTexture	World{:World}	[#texture: "WorldWaves{:Setting}_Texture", #renderFormat: #rgba8888]
	SetShader	World{:World}	[#shader: "WorldCitySea_Material", #texture1: "WorldSea{:Setting}_Texture"]
	SetShader	World{:World}	[#shader: "WorldCityWaves_Material", #texture1: "WorldWaves{:Setting}_Texture"]	
	
	Evaluate	"{:Setting}"="Night"	[#event: "WorldCityAdditiveTextures"]	
	
	// Skybox
	Event	SkyboxInitialization
	SetSkybox	World{:World}	[#skybox: "SkyboxCity"]	
SetupWorldCity_End

LandMarkCity_Begin
	PutMessage	--> LandMarkCity
	// Sound
	PlaySound	SpecialPickUp_SFX	[#override: TRUE]
	// Text textures
	CreateTextTexture	Interface	[#tag: #InterfaceStatusHeaderText, #style: #StatusTitle, #text: #LandmarkInformationTitle]
	CreateTextTexture	Interface	[#tag: #InterfaceStatusBodyText, #style: #StatusBody, #text: #LandmarkInformation]
	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World"]
	// Delay next event
	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
	// Do it
	SetModel	World{:World}	[#model: "02_07_landmarkcity_1_object", #resource: "landmarkcitydestroyed_object", #shader: "WorldCityMisc_Material"] 
	AddToList	gGame.UPLandMarks	[#value: "LandMarkCity_Object", #override: TRUE]
	SaveData	x	x
	CheckAllLandMarks	x	x
LandMarkCity_End
