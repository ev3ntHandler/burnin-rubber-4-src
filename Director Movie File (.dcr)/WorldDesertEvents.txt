SetupWorldDesert_Begin
	PutMessage	--> SetupWorldDesert
	// ModelList
	SetValue		[#WorldModelList: {:WorldModels}.duplicate()]
	AddToList	gGame.WorldModelList	[#value: "Sea"]
	// Additional textures & shader sets
	Evaluate	"{:Setting}"="Night"	[#Event: "WorldDesertAdditiveTextures"]	
	CreateTexture	World{:World}	[#texture: "WorldSea{:Setting}_Texture"]
	SetShader	World{:World}	[#shader: "WorldDesertSea_Material", #texture1: "WorldSea{:Setting}_Texture"]
	// Skybox
	Event	SkyboxInitialization
	SetSkybox	World{:World}	[#skybox: "SkyboxDesert"]	
SetupWorldDesert_End

SetupWorldDesertSpecialEffects_Begin
	PutMessage	SetupWorldDesertSpecialEffects	
	AddToScroller	World{:World}	[#shader: "WorldDesertSea_Material", #position: vector(-0.001, -0.001, 0.0)]	
SetupWorldDesertSpecialEffects_End

LandMarkDesert_Begin
	PutMessage	--> LandMarkDesert
	// Sound
	PlaySound	SpecialPickUp_SFX	[#override: TRUE]
	// Text textures
	CreateTextTexture	Interface	[#tag: #InterfaceStatusHeaderText, #style: #StatusTitle, #text: #LandmarkInformationTitle]
	CreateTextTexture	Interface	[#tag: #InterfaceStatusBodyText, #style: #StatusBody, #text: #LandmarkInformation]
	// Show it
	SetVisible	Interface	[#model: "InterfaceStatus", #state: TRUE, #parent: "World{:World}"]
	// Delay next event
	DelayEvent		[#delay: {:StatusDelay}, #name: "HideStatus", #event: "HideStatus"]
	// Do it
	SetModel	World{:World}	[#model: "07_03_landmarkdesert_1_object", #resource: "landmarkdesertdestroyed_object", #shader: "WorldDesertMisc_Material"] 
	AddToList	gGame.UPLandMarks	[#value: "LandMarkDesert_Object", #override: TRUE]
	SaveData	x	x
	CheckAllLandMarks	x	x
LandMarkDesert_End
