SetupTrack_Begin
	PutMessage	SetupTrack
	//-----------------------------------------------------------------------------
	// Start showing init update
	//-----------------------------------------------------------------------------
	SetValue		[#ForceUpdate: TRUE]
	SetValue		[#ForceState: #initializing]
	UpdateTagsInitializingGame	x	x
	//-----------------------------------------------------------------------------
	// Remove from new List
	//-----------------------------------------------------------------------------
	RemoveFromNew	x	x
	//-----------------------------------------------------------------------------
	// Scene!
	//-----------------------------------------------------------------------------
	SetValue		[#Scene: "Track"]
	//-----------------------------------------------------------------------------
	// Update track info & data & SW Trophies
	//-----------------------------------------------------------------------------
	UpdateTrackInfo	x	x
	GetChallengeType	x	x
	GetTicket	x	x
	//-----------------------------------------------------------------------------
	// Setup challenge (do this first as it can adjust AI amount and such)
	//-----------------------------------------------------------------------------
	SetValue		[#Assets: gGame.TrackInfo.Assets]
	SetValue		[#Challenge: gGame.TrackInfo.GameMode]
	SetValue		[#AIFriendlyFire: gGame.TrackInfo.AIFriendlyFire]
	SetValue		[#Vehicle: gGame.TrackInfo.Vehicle]
	UpdateVehicleType	x	x
	//-----------------------------------------------------------------------------
	// Build sprite & add values
	//-----------------------------------------------------------------------------
	CreateSprite	{:Assets}Assets	[#channel: 1, #default3dScripts: TRUE]
	AddValue		[#name: "Member", #value: member("{:Assets}Assets")]
	AddValue		[#name: "Sprite", #value: sprite(1)]
	//-----------------------------------------------------------------------------
	// Setup objects, pickups & debris
	//-----------------------------------------------------------------------------
	Event	SetupObjects
	Event	SetupDebris
	Event	SetupPickups	
	//-----------------------------------------------------------------------------
	// Clone objects, pickups and debris
	//-----------------------------------------------------------------------------
	CloneModel	{:Assets}Assets	[#SourceMember: "Objects", #lod: FALSE, #delete: TRUE]
	CloneModel	{:Assets}Assets	[#SourceMember: "Debris",   #lod: FALSE, #delete: TRUE]
	CloneModel	{:Assets}Assets	[#SourceMember: "Pickups", #lod: FALSE, #delete: TRUE]
	//-----------------------------------------------------------------------------
	// Init specific challenge
	//-----------------------------------------------------------------------------
	Event	Setup{:Challenge}Challenge
	//-----------------------------------------------------------------------------
	// Build track & skybox
	//-----------------------------------------------------------------------------
	Event	{:Assets}Textures
	Evaluate	{:DailyChallenge}=TRUE	[#event: gGame.TrackInfo.Reference&"Build"]
	Evaluate	{:DailyChallenge}=FALSE	[#event: "{:Track}Build"]
	Event	SkyboxInitialization
	//-----------------------------------------------------------------------------
	// Scripts
	//-----------------------------------------------------------------------------
	Event	ChallengeScripts
	//-----------------------------------------------------------------------------
	// Ambient style & reset some shaders
	//-----------------------------------------------------------------------------
	SetAmbientStyle	{:Assets}Assets	[#style: gGame.trackinfo.ambientStyle]
	SetShader	{:Assets}Assets	[#shader: "Pickups_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "Arrows_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "Crosses_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "Finish_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "Decals_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "ImpactSmoke_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "ImpactBurst_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "ImpactSparks_Material", #emissive: rgb(255,255,255)]
	SetShader	{:Assets}Assets	[#shader: "ImpactDebris_Material", #emissive: rgb(128,128,128)]
	//-----------------------------------------------------------------------------
	// Effects (skybox, scroller, particles, bloom, etc.)
	//-----------------------------------------------------------------------------
	SetSkybox	{:Assets}Assets	[#skybox: gGame.TrackInfo.Skybox]	
	AddToScroller	{:Assets}Assets	[#shader: "Arrows_Material", #time: 150, #position: vector(0.25, 0.0, 0.0)]
	AddToScroller	{:Assets}Assets	[#shader: "Finish_Material", #time: 30, #position: vector(-0.01, 0.0, 0.0)]
	AddScript	#System	[#script: "[PS] Motion Blur", #data: [#member: "{:Assets}Assets", #camera: "Player1_Camera", #passes: 7]]
	AddScript	#System	[#script: "[PS] Bloom", #data: [#member: "{:Assets}Assets", #camera: "Player1_Camera", #template: gGame.TrackInfo.Bloom, #skybox: TRUE]]
	//-----------------------------------------------------------------------------
	// Overlay hack
	//-----------------------------------------------------------------------------
	SetVisible	Interface	[#model: "InterfaceOverlay", #state: TRUE, #parent: "World"]
	//-----------------------------------------------------------------------------
	// Starttime
	//-----------------------------------------------------------------------------
	SetStartTime	x	x	
	//-----------------------------------------------------------------------------
	// TEMP: CHEATS	
	//-----------------------------------------------------------------------------
	//UnlockAllTickets	x	x
	//WinCurrentTickets	x	x
	//-----------------------------------------------------------------------------
	// Fade to show challenge
	//-----------------------------------------------------------------------------
	Fade	{:Assets}Assets	[#hold: 500, #time: 750, #type: #out, #event: "DelayShowChallenge"]
	//-----------------------------------------------------------------------------
	// show sprite
	//-----------------------------------------------------------------------------
	SetSprite	{:Sprite}
	SetValue		[#ForceUpdate: FALSE]
	//Event	CountDownGo	
SetupTrack_End

ChallengeScripts_Begin
	PutMessage	--> ChallengeScripts
	// Player(s)
	Iterate	{:Players}	[#event: "ChallengePlayer{:Vehicle}Scripts"]
	Evaluate	#{:Effects}=#low	[#handler: "TurnOffVehicleEffects"]
	// AI(s)
	SetValue		[#AI: gGame.TrackInfo.Enemies.count]
	//SetValue		[#AI: 0]
	SetValue		[#AIVehicle: "{:Vehicle}"]
	Evaluate	"{:Track}"="SpecialMissionConvoy"	[#event: "AIVehicleSpecialMissionConvoy"]
	Evaluate	"{:Track}"="BossEnd"	[#event: "AIVehicleBossEnd"]	
	Iterate	{:AI}	[#event: "ChallengeAI{:AIVehicle}Scripts"]
	// Camera
	AddScript		[#script: "[PS] Camera", #data: [#member: "{:Assets}Assets", #target: gGame.Vehicles[1].general.tag]]
	SetFog	World{:World}	[#camera: "Player1_Camera", #color: gGame.TrackInfo.FogColor, #range: gGame.TrackInfo.FogRange]
	// UpdateCamera
	SetValue	gGame.Camera[1]	[#DefaultSet: symbol(gGame[#{:Vehicle}])]
	UpdateCameraSet	gGame[#{:Vehicle}]	
	// Interface
	Event	SetupInterface		
ChallengeScripts_End
